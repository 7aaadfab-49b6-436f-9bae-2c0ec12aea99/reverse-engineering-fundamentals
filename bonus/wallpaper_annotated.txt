================================================================================
 BLOCK 2: read your input

40009d: lea  0x400297,%rsi    ; rsi = address of an empty buffer that will hold your input

================================================================================
 BLOCK 3: loop #1 -- scans your input BACKWARD, one byte at a time
================================================================================
4000ae: lea  0x400297,%rbx        ; rbx = base address of your input buffer (fixed for the rest of the program)
4000b6: mov  $0x26,%rcx           ; rcx = 38, used here purely as a countdown index

    LOOP #1 START  (walks index rcx from 37 down to 0) --------------------
    |
    |   4000bd: dec  %rcx                     ; rcx = rcx - 1  (first pass: 38 -> 37)
    |   4000c0: js   0x4000dd                 ; if rcx just went negative (i.e. rcx WAS 0 and is now -1):
    |                                         ;    every index has been checked -> LEAVE loop #1, fall into BLOCK 4
    |
    |   4000c2: movzbq (%rbx,%rcx,1),%rax     ; rax = the single byte at input[rcx], zero-extended to 64 bits
    |   4000c7: movabs $0xf000000000400,%rdx  ; rdx = a fixed 64-bit constant -- treat it as a 64-bit BITMASK,
    |                                         ;   one bit per possible byte VALUE 0..63 (only the low 6 bits of
    |                                         ;   rax actually select which bit of rdx gets tested)
    |   4000d1: bt   %rax,%rdx                ; CF (carry flag) = bit number (rax mod 64) of rdx
    |                                         ;   in plain words: "is input[rcx]'s value one of the values
    |                                         ;   this mask marks as allowed?"
    |   4000d5: jae  0x400226                 ; jae == "jump if CF was 0" == that bit was NOT set
    |                                         ;   -> this byte failed the allow-list check -> BLOCK 8's
    |                                         ;      "wrong password" branch, program ends here
    |   4000db: jmp  0x4000bd                 ; bit WAS set, byte is allowed -> go back to top, check next index
    |
    LOOP #1 END  (only reached by the js at 4000c0 falling through to 4000dd) -

 BLOCK 4 / BLOCK 5 / BLOCK 6: loop #2 -- the main per-character scrambler

    nested inside the same iteration as block 4, and control returns to
    the top of block 4 every time block 6 finishes -- see 4001d4)

4000dd: movabs $0xb6fd071e9c8a3425,%rax   ; rax = the running "accumulator" -- this is the ONE value that

|
|   ---- exit test for the whole outer loop -------------------------------
|   4000ea: movabs $0x2000000000,%rsi      ; rsi = a fixed constant used only to encode "which index is the
|                                            ;   last character" -- exactly one bit of it is set
|   4000f4: bt   %rdi,%rsi                 ; CF = bit number (rdi mod 64) of rsi
|   4000f8: jb   0x4001d9                  ; if that bit IS set: rdi has reached the stopping index ->
|                                            ;   LEAVE loop #2 entirely, jump straight to BLOCK 7 (final check)
|                                            ;   (note: the length loop #1 walks (38) and the index that stops
|                                            ;   loop #2 are NOT necessarily the same number)

|   ---- load this character -----------------------------------------------

|   400101: inc  %rcx                      ; rcx enters this pass as 0xffffffffffffffff (set at 4001ce/4001d1
|                                            ;   on the previous pass, or implicitly -1 the first time), so
|                                            ;   this inc makes rcx = 0
|   400104: shl  $0x2,%rcx                 ; rcx = rcx * 4 -- rcx is being used below as a ROTATE-AMOUNT
|                                            ;   counter, always kept a multiple of 4

|   |                      of times per character, "peeling" the
|   |                      accumulator 4 bits at a time until it hits a

|   |   400108: ror  %cl,%rax              ; rotate accumulator RIGHT by rcx bits (only %cl, the low byte
|   |                                        ;   of rcx, is used as the rotate amount)
|   |   40010b: mov  %rax,%rsi             ; rsi = snapshot of the just-rotated accumulator
|   |   40010e: and  $0xf,%sil             ; sil = lowest 4 bits (one nibble) of that snapshot
|   |   400112: push %rsi                  ; push the whole qword (top byte = sil) onto the stack --
|   |                                        ;   from here the value is manipulated IN MEMORY, on the stack,
|   |                                        ;   not in a register

|   |   400116: or   (%rsp),%sil           ; sil = sil OR (that negated byte)
|   |                                        ;   -- classic "was the nibble zero?" trick: n | (-n) is 0 ONLY
|   |                                        ;   when n itself was already 0; for any nonzero n it is nonzero
|   |   40011a: je   0x400128              ; if the result was zero (nibble WAS zero) -> BREAK the inner
|   |                                        ;   sub-loop, fall through past it
|   |
|   |   40011c: rol  %cl,%rax              ; nibble was NON-zero: undo the ror from 400108 (rotate back left)
|   |   40011f: shr  $0x2,%rcx             ; rcx = rcx / 4  (undo the earlier *4 scaling, one "step" worth)
|   |   400123: inc  %rcx                  ; rcx += 1        (advance to the next rotate amount)
|   |   400126: jmp  0x400104              ; back to the top of THIS inner sub-loop with the new rcx
|   |
|   INNER SUB-LOOP END  (only reached via the je at 40011a) -----------------
|
|   400128: rol  %cl,%rax                  ; undo the final ror once more (restores accumulator's orientation
|                                            ;   to match how many steps the inner sub-loop actually took)
|   40012b: shr  $0x2,%rcx                 ; rcx /= 4 -- rcx now records HOW MANY sub-loop steps happened
|                                            ;   before a zero nibble was found
|   40012f: movzbq %dl,%rdx                ; rdx = your character's raw byte value, zero-extended
|   400133: add  $0x8,%rsp                 ; pop the pushed qword off the stack (without reading it into a reg)
|   400137: movabs $0x3bb97ffd7ffd6eec,%rsi ; rsi = another fixed 64-bit constant, used as a second bitmask
|   400141: shl  $0x2,%rcx                 ; rcx *= 4 again
|   400145: add  $0xd0,%dl                 ; dl = your character's value + 0xd0 (wraps mod 256)
|   400148: add  %rdx,%rcx                 ; rcx = rcx + (that shifted character value)
|   40014b: bt   %rcx,%rsi                 ; CF = bit number (rcx mod 64) of rsi
|   40014f: jae  0x400226                  ; that bit NOT set -> this character fails -> "wrong password"
|                                            ;   (BLOCK 8), program ends here
|
|   400155: sub  %rdx,%rcx                 ; rcx = rcx - (character value + 0xd0)  -- undoes the add at 400148

|

|   BLOCK 5 (nested):  4-way jump table -- picks exactly ONE of 4 fixed

|                       all 4 paths rejoin at the SAME place (400187)

|   |
|   |   400160: call 0x400165        ; "get my own instruction pointer" trick -- NOT a real function call,
|   |                                  ;   this just pushes the address 0x400165 onto the stack so it can be
|   |                                  ;   popped into a register on the next line
|   |   400165: pop  %rsi            ; rsi = 0x400165 (the address right after this call)
|   |   400166: add  %rdx,%rsi       ; rsi = 0x400165 + rdx  -> the actual jump target, chosen by rdx
|   |   400169: jmp  *%rsi           ; jump to whichever of the 4 addresses below rdx pointed at
|   |
|   |       400016b: jmp 0x400173    ; \
|   |       40016d: jmp 0x40017a     ;  \  four tiny trampolines, one per possible rdx value,
|   |       40016f: jmp 0x400181     ;  /  each landing on a different "push $constant" below
|   |       400171: jmp 0x400185     ; /
|   |
|   |       PATH A -> 400173: push $0xf0    ; \
|   |       PATH B -> 40017a: push $0xfc    ;  \  exactly ONE of these four executes per character,
|   |       PATH C -> 400181: push $0x10    ;  /  pushing one fixed byte-ish constant onto the stack
|   |       PATH D -> 400185: push $0x4     ; /
|   |
|   |       (all four paths fall straight through into BLOCK 6 at 400187 --
|   |        there is no separate "join" instruction, they just converge
|   |        by flowing into the same next address)
|   |
|   ------------------------------------------------------------------------
|   BLOCK 6 (nested):  further scrambling using whichever constant BLOCK 5
|                       picked, operating on a value sitting ON THE STACK

|   ------------------------------------------------------------------------
|   |

|   |                                       ;   [top] path-constant, accumulator-copy, ... )

|   |   40018c: rorq %cl,(%rsp)           ; rotate the QWORD *sitting in memory at (%rsp)* right by cl bits
|   |                                       ;   (note: this rotates a stack SLOT, not a register)

|   |   4001af: sub  0x8(%rsp),%cl        ; cl -= the path-constant again
|   |   4001b3: rorq %cl,(%rsp)           ; rotate the stack slot right by that adjusted amount
|   |   4001b7: or   %sil,(%rsp)          ; OR the saved low nibble (from 400199) back into the stack byte
|   |   4001bb: neg  %rcx                 ; rcx = -rcx
|   |   4001be: add  $0x40,%rcx           ; rcx = 64 - rcx
|   |   4001c2: rorq %cl,(%rsp)           ; one final rotate of the stack slot by that amount
|   |   4001c6: pop  %rax                 ; rax = the fully re-scrambled accumulator, pulled back off the stack
|   |   4001c7: add  $0x8,%rsp            ; discard the leftover path-constant slot underneath it

|   ---- advance to next character, reset rcx, loop back --------------------
|   4001cb: inc  %rdi                     ; rdi += 1  -- move on to the next character of your input
|   4001ce: xor  %rcx,%rcx                ; rcx = 0

================================================================================
 NOTE on the success message
================================================================================
The success path prints your OWN input back inside "CMO{...}". Nothing in
this binary stores the correct password anywhere as plaintext -- there is no
`cmp` against a fixed string. "Correctness" is entirely emergent from whether
your bytes drive the accumulator in BLOCK 4/5/6 to exactly zero after the
three XORs in BLOCK 7. That means there is no single address you could jump
to that "has" the password sitting in memory -- the checks (loop #1's
allow-list, loop #2's per-character bit tests, and the final zero test) are
gates on an algorithm, not lookups against stored data.
